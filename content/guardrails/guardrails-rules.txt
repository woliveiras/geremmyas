# Guardrails Rules
#
# Format: ACTION PATTERN
#   ACTION = BLOCK | ASK | ALLOW
#   PATTERN = regex matched against the command string
#
# BLOCK: reject immediately, show reason
# ASK:   request authority only at a protected boundary
# ALLOW: accept a command carrying explicit verified-target context
#
# Lines starting with # are comments. Blank lines are ignored.

# --- Destructive version control ---
BLOCK ^[[:space:]]*git([[:space:]]+-C[[:space:]]+[^[:space:]]+)?[[:space:]]+push.*((--force[^[:space:]]*|-f)([[:space:]]|$)|[[:space:]]+\+[^[:space:]]+)
ASK   ^[[:space:]]*git([[:space:]]+-C[[:space:]]+[^[:space:]]+)?[[:space:]]+push
BLOCK ^[[:space:]]*git([[:space:]]+-C[[:space:]]+[^[:space:]]+)?[[:space:]]+reset.*[[:space:]]+--hard([[:space:]]|$)
BLOCK ^[[:space:]]*git([[:space:]]+-C[[:space:]]+[^[:space:]]+)?[[:space:]]+clean([[:space:]]|$)
BLOCK ^[[:space:]]*git([[:space:]]+-C[[:space:]]+[^[:space:]]+)?[[:space:]]+checkout[[:space:]]+--[[:space:]]+\.

# --- Destructive file operations ---
BLOCK ^[[:space:]]*rm([[:space:]]+--?[[:alpha:]-]+)*[[:space:]]+(--[[:space:]]+)?(/|~|\$HOME)[[:space:]]*$

# --- Infrastructure ---
BLOCK ^[[:space:]]*(GEREMMYAS_TARGET=(local|test|disposable)[[:space:]]+)?terraform.*[[:space:]]destroy([[:space:]]|$)
ALLOW ^[[:space:]]*GEREMMYAS_TARGET=(local|test|disposable)[[:space:]]+terraform([[:space:]]+-chdir=[^[:space:]]+)?[[:space:]]+(apply|import|state[[:space:]]+(rm|mv))([[:space:]]|$)
ASK   ^[[:space:]]*terraform([[:space:]]+-chdir=[^[:space:]]+)?[[:space:]]+(apply|import|state[[:space:]]+(rm|mv))([[:space:]]|$)

# --- Cloud and database mutations ---
ALLOW ^[[:space:]]*GEREMMYAS_TARGET=(local|test|disposable)[[:space:]]+(gcloud|psql)([[:space:]]|$)
ASK   ^[[:space:]]*gcloud.*[[:space:]](create|update|delete|deploy|enable|disable|add-iam-policy-binding|remove-iam-policy-binding)([[:space:]]|$)
ASK   ^[[:space:]]*psql.*(DROP[[:space:]]+(TABLE|DATABASE)|TRUNCATE|DELETE[[:space:]]+FROM)

# --- Privileged system mutation ---
ASK   sudo

# --- Secrets ---
BLOCK curl.*Authorization.*Bearer
BLOCK echo.*SECRET
BLOCK echo.*TOKEN
BLOCK echo.*PASSWORD
